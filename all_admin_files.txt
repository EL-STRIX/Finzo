<?php
/**
 * Finzo Banking System — Admin Audit Log API
 * GET endpoint: paginated audit log entries.
 * Version 1.0 — April 2026
 */

require_once dirname(__DIR__, 3) . '/middleware/auth-admin.php';
if ($_SERVER['REQUEST_METHOD'] !== 'GET') {
    json_response(['success' => false, 'message' => 'Method not allowed'], 405);
}

try {
    $search      = clean_input($_GET['search'] ?? '');
    $action_type = clean_input($_GET['action_type'] ?? 'all');
    $actor_role  = clean_input($_GET['actor_role'] ?? 'all');
    $from_date   = clean_input($_GET['from_date'] ?? '');
    $to_date     = clean_input($_GET['to_date'] ?? '');
    $page        = max(1, intval($_GET['page'] ?? 1));
    $per_page    = min(100, max(10, intval($_GET['per_page'] ?? 25)));

    $where_clauses = ["1=1"];
    $params = [];

    if ($action_type !== 'all') {
        $where_clauses[] = "al.action_type LIKE ?";
        $params[] = "%{$action_type}%";
    }

    if ($actor_role !== 'all' && in_array($actor_role, ['admin', 'staff', 'system', 'user'])) {
        if ($actor_role === 'admin') {
            $where_clauses[] = "(al.actor_role LIKE '%Admin%' OR al.actor_role = 'Moderator' OR al.actor_role IN ('admin', 'super_admin', 'moderator'))";
        } else if ($actor_role === 'staff') {
            $where_clauses[] = "al.actor_role IN ('Staff', 'staff')";
        } else if ($actor_role === 'system') {
            $where_clauses[] = "al.actor_role IN ('System', 'system')";
        } else if ($actor_role === 'user') {
            $where_clauses[] = "al.actor_role IN ('User', 'user')";
        }
    }

    if (!empty($from_date)) {
        $where_clauses[] = "DATE(al.created_at) >= ?";
        $params[] = $from_date;
    }

    if (!empty($to_date)) {
        $where_clauses[] = "DATE(al.created_at) <= ?";
        $params[] = $to_date;
    }

    if (!empty($search)) {
        $where_clauses[] = "(al.actor_id LIKE ? OR al.entity_id LIKE ?)";
        $s = "%{$search}%";
        $params[] = $s;
        $params[] = $s;
    }

    $where_sql = implode(' AND ', $where_clauses);

    // Count
    $stmt = $pdo->prepare("SELECT COUNT(*) FROM audit_log al WHERE {$where_sql}");
    $stmt->execute($params);
    $total = (int)$stmt->fetchColumn();

    $pages  = max(1, ceil($total / $per_page));
    $offset = ($page - 1) * $per_page;

    $stmt = $pdo->prepare("
        SELECT al.id, al.action_type, al.entity_id, al.entity_type,
               al.actor_id, al.actor_role, al.details, al.ip_address, al.created_at,
               al.actor_id as actor_display_id,
               al.actor_role as actor_specific_role,
               CASE 
                   WHEN al.entity_type = 'user' THEN COALESCE((SELECT customer_id FROM users WHERE id = al.entity_id OR customer_id = al.entity_id LIMIT 1), al.entity_id)
                   WHEN al.entity_type = 'staff' THEN COALESCE((SELECT staff_id FROM staff WHERE id = al.entity_id OR staff_id = al.entity_id LIMIT 1), al.entity_id)
                   WHEN al.entity_type = 'admin' THEN COALESCE((SELECT admin_id FROM admin_users WHERE id = al.entity_id OR admin_id = al.entity_id LIMIT 1), al.entity_id)
                   ELSE al.entity_id 
               END as entity_display_id
        FROM audit_log al
        WHERE {$where_sql}
        ORDER BY al.created_at DESC
        LIMIT {$per_page} OFFSET {$offset}
    ");
    $stmt->execute($params);
    $entries = $stmt->fetchAll();

    // Get distinct action types for filter dropdown
    $stmt = $pdo->query("SELECT DISTINCT action_type FROM audit_log ORDER BY action_type");
    $action_types = $stmt->fetchAll(PDO::FETCH_COLUMN);

    // Compute exact counts based on current filters
    $stmt_stats = $pdo->prepare("
        SELECT 
            SUM(CASE WHEN al.actor_role LIKE '%Admin%' OR al.actor_role = 'Moderator' OR al.actor_role IN ('admin', 'super_admin', 'moderator') THEN 1 ELSE 0 END) as admin_count,
            SUM(CASE WHEN al.actor_role IN ('Staff', 'staff') THEN 1 ELSE 0 END) as staff_count,
            SUM(CASE WHEN al.actor_role IN ('User', 'user') THEN 1 ELSE 0 END) as user_count
        FROM audit_log al
        WHERE {$where_sql}
    ");
    $stmt_stats->execute($params);
    $stats = $stmt_stats->fetch();

    json_response([
        'success'      => true,
        'entries'      => $entries,
        'total'        => $total,
        'pages'        => $pages,
        'current_page' => $page,
        'per_page'     => $per_page,
        'action_types' => $action_types,
        'stats'        => [
            'admin'  => (int)$stats['admin_count'],
            'staff'  => (int)$stats['staff_count'],
            'user'   => (int)$stats['user_count']
        ]
    ]);

} catch (PDOException $e) {
    error_log("Audit log API error: " . $e->getMessage());
    json_response(['success' => false, 'message' => 'Failed to load audit log'], 500);
}

if ($_SERVER['REQUEST_METHOD'] !== 'GET') {
    json_response(['success' => false, 'message' => 'Method not allowed'], 405);
}

try {
    // 1. Total users
    $stmt = $pdo->query("SELECT COUNT(*) FROM users WHERE is_deleted = 0");
    $total_users = (int)$stmt->fetchColumn();

    // 2. Active users
    $stmt = $pdo->query("SELECT COUNT(*) FROM users WHERE status = 'active' AND is_deleted = 0");
    $active_users = (int)$stmt->fetchColumn();

    // 3. Pending users (pending + under_review)
    $stmt = $pdo->query("SELECT COUNT(*) FROM users WHERE status IN ('pending', 'under_review') AND is_deleted = 0");
    $pending_users = (int)$stmt->fetchColumn();

    // 4. Suspended users
    $stmt = $pdo->query("SELECT COUNT(*) FROM users WHERE status = 'suspended' AND is_deleted = 0");
    $suspended_users = (int)$stmt->fetchColumn();

    // 5. Total staff
    $stmt = $pdo->query("SELECT COUNT(*) FROM staff");
    $total_staff = (int)$stmt->fetchColumn();

    // 6. Total branches
    $stmt = $pdo->query("SELECT COUNT(*) FROM branches");
    $total_branches = (int)$stmt->fetchColumn();

    // 7. Transactions today
    $stmt = $pdo->query("SELECT COUNT(*) FROM transactions WHERE DATE(created_at) = CURDATE()");
    $transactions_today = (int)$stmt->fetchColumn();

    // 8. Transactions this month
    $stmt = $pdo->query("SELECT COUNT(*) FROM transactions WHERE MONTH(created_at) = MONTH(CURDATE()) AND YEAR(created_at) = YEAR(CURDATE())");
    $transactions_month = (int)$stmt->fetchColumn();

    // 9. Deposits today
    $stmt = $pdo->query("SELECT COALESCE(SUM(amount), 0) FROM transactions WHERE type = 'deposit' AND status = 'completed' AND DATE(created_at) = CURDATE()");
    $deposits_today = (float)$stmt->fetchColumn();

    // 10. Withdrawals today
    $stmt = $pdo->query("SELECT COALESCE(SUM(amount), 0) FROM transactions WHERE type = 'withdrawal' AND status = 'completed' AND DATE(created_at) = CURDATE()");
    $withdrawals_today = (float)$stmt->fetchColumn();

    // 11. Transfers today
    $stmt = $pdo->query("SELECT COALESCE(SUM(amount), 0) FROM transactions WHERE type = 'transfer' AND status = 'completed' AND DATE(created_at) = CURDATE()");
    $transfers_today = (float)$stmt->fetchColumn();

    // 12. Pending approvals (pending + under_review)
    $stmt = $pdo->query("SELECT COUNT(*) FROM users WHERE status IN ('pending', 'under_review') AND is_deleted = 0");
    $pending_approvals = (int)$stmt->fetchColumn();

    // 13. Locked accounts
    $stmt = $pdo->query("SELECT COUNT(*) FROM users WHERE locked_until > NOW() AND is_deleted = 0");
    $locked_accounts = (int)$stmt->fetchColumn();

    // 13a. Total accounts
    $stmt = $pdo->query("SELECT COUNT(*) FROM accounts WHERE status = 'active'");
    $total_accounts = (int)$stmt->fetchColumn();

    // 13b. Interest credited
    $stmt = $pdo->query("SELECT COALESCE(SUM(amount), 0) FROM transactions WHERE type = 'interest_credit' AND status = 'completed'");
    $interest_credited = (float)$stmt->fetchColumn();

    // 13c. Soft deleted users
    $stmt = $pdo->query("SELECT COUNT(*) FROM users WHERE is_deleted = 1");
    $soft_deleted_users = (int)$stmt->fetchColumn();

    // 14. Recent users (last 10) with profile picture
    $stmt = $pdo->query("
        SELECT id, full_name, email, status, created_at, profile_picture
        FROM users
        WHERE is_deleted = 0
        ORDER BY created_at DESC
        LIMIT 10
    ");
    $recent_users = $stmt->fetchAll();

    foreach ($recent_users as &$u) {
        $profile_pic_url = $u['profile_picture'] ? APP_URL . '/' . $u['profile_picture'] : null;

        $u['profile_picture_url'] = $profile_pic_url;

        $month_label = date('M Y', strtotime("-{$i} months"));

            SELECT
                COALESCE(SUM(CASE WHEN type = 'deposit' THEN amount ELSE 0 END), 0) as deposits,
                COALESCE(SUM(CASE WHEN type = 'withdrawal' THEN amount ELSE 0 END), 0) as withdrawals,
                COALESCE(SUM(CASE WHEN type = 'transfer' THEN amount ELSE 0 END), 0) as transfers,
                COUNT(*) as count
            FROM transactions
            WHERE status = 'completed'
                AND DATE_FORMAT(created_at, '%Y-%m') = ?
        ");
        $stmt->execute([$month_date]);
        $month_data = $stmt->fetch();

        $monthly_chart_data[] = [
            'month'       => $month_label,
            'deposits'    => (float)$month_data['deposits'],
            'withdrawals' => (float)$month_data['withdrawals'],
            'transfers'   => (float)$month_data['transfers'],
            'count'       => (int)$month_data['count']
        ];
    }

    // Total system balance
    $stmt = $pdo->query("SELECT COALESCE(SUM(balance), 0) FROM accounts WHERE status = 'active'");
    $total_balance = (float)$stmt->fetchColumn();

    // Fetch fresh admin data
    $stmt = $pdo->prepare("SELECT full_name, email, profile_photo, admin_role FROM admin_users WHERE id = ?");
    $stmt->execute([$_SESSION['admin_id']]);
    $admin_data = $stmt->fetch();

    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }

    json_response([
        'success' => true,
        'csrf_token' => $_SESSION['csrf_token'],
        'admin' => [
            'id' => $_SESSION['admin_id'],
            'full_name' => $admin_data['full_name'],
            'email' => $admin_data['email'],
            'profile_photo' => $admin_data['profile_photo'],
            'role' => $admin_data['admin_role']
        ],
        'stats' => [
            'total_users'        => $total_users,

            'suspended_users'    => $suspended_users,

        SELECT COUNT(*) 
        FROM users u 
        LEFT JOIN accounts a ON u.customer_id = a.user_id 
        WHERE {$where_sql}
    ";
    $stmt = $pdo->prepare($count_sql);
    $stmt->execute($params);
    $total = (int)$stmt->fetchColumn();

    $pages  = max(1, ceil($total / $per_page));

               a.account_no, a.account_type, a.balance

<?php
/**
 * Finzo Banking System — Admin Users CSV Export API
 * GET endpoint: Generates and downloads a CSV of users based on current filters.
 * Version 1.0 — April 2026
 */

require_once dirname(__DIR__, 3) . '/middleware/auth-admin.php';

if ($_SERVER['REQUEST_METHOD'] !== 'GET') {
    json_response(['success' => false, 'message' => 'Method not allowed'], 405);
}

try {
    $search       = clean_input($_GET['search'] ?? '');
    $status       = clean_input($_GET['status'] ?? 'all');

    // Build query
    $where_clauses = ["u.is_deleted = 0", "u.status != 'suspended'"];
    $params = [];

    // Status filter
    if ($status !== 'all' && in_array($status, ['pending', 'under_review', 'active', 'rejected'])) {
        $where_clauses[] = "u.status = ?";
        $params[] = $status;
    }

    // Search
    if (!empty($search)) {
        $where_clauses[] = "(u.full_name LIKE ? OR u.customer_id LIKE ?)";
        $search_param = "%{$search}%";
        $params[] = $search_param;
        $params[] = $search_param;
    }

    $where_sql = implode(' AND ', $where_clauses);

    // Fetch users for export (NO pagination)
    $sql = "
        SELECT
            u.customer_id, u.full_name, u.email, u.phone, u.status, u.created_at, u.last_login_at
        FROM users u
        WHERE {$where_sql}
        ORDER BY u.created_at DESC
    ";

    $stmt = $pdo->prepare($sql);
    $stmt->execute($params);
    
    // Set headers for CSV download
    header('Content-Type: text/csv; charset=utf-8');
    header('Content-Disposition: attachment; filename=Finzo_Users_Export_' . date('Y-m-d') . '.csv');
    
    // Open output stream
    $output = fopen('php://output', 'w');
    
    // Output column headings
    fputcsv($output, ['Customer ID', 'Full Name', 'Email', 'Phone', 'Status', 'Joined Date', 'Last Active']);
    
    // Output data rows
    while ($row = $stmt->fetch(PDO::FETCH_ASSOC)) {
        fputcsv($output, [
            $row['customer_id'] ?: 'Pending',
            $row['full_name'],
            $row['email'],
            $row['phone'],
            ucfirst($row['status']),
            $row['created_at'],
            $row['last_login_at'] ?: 'Never'
        ]);
    }
    
    fclose($output);
    exit();

} catch (PDOException $e) {
    error_log("Admin users export API error: " . $e->getMessage());
    die("Failed to generate export file.");
}

<?php
/**
 * Finzo Banking System — Admin Interest Rate Management API
 * GET: fetch current rate and history | POST: update rate
 * Version 1.0 — April 2026
 */

require_once dirname(__DIR__, 3) . '/middleware/auth-admin.php';

$admin_id = $_SESSION['admin_id'];

// =====================================================
// GET: CURRENT RATE + HISTORY
// =====================================================
if ($_SERVER['REQUEST_METHOD'] === 'GET') {
    try {
        // Current rate (most recent)
        $stmt = $pdo->query("
            SELECT ir.rate, ir.apply_from_type, ir.apply_from_date, ir.created_at,
                   au.full_name as changed_by_name
            FROM interest_rates ir
            JOIN admin_users au ON ir.changed_by_admin_id = au.admin_id
            ORDER BY ir.created_at DESC
            LIMIT 1
        ");
        $current_rate_row = $stmt->fetch();
        
        $current_rate = $current_rate_row ? (float)$current_rate_row['rate'] : 4.25;
        $last_change_date = $current_rate_row ? date('d M Y', strtotime($current_rate_row['created_at'])) : '-';

        // Stats
        $stats = [
            'total_interest_credited' => 0,
            'interest_delta' => '+0%',
            'earning_accounts' => 0
        ];
        
        $stmt = $pdo->query("SELECT SUM(amount) FROM transactions WHERE type = 'interest_credit' AND status = 'completed'");
        $total_interest = (float)$stmt->fetchColumn();
        $stats['total_interest_credited'] = '₹' . number_format($total_interest, 2);

        $stmt = $pdo->query("SELECT COUNT(*) FROM accounts WHERE account_type = 'savings' AND status = 'active'");
        $stats['earning_accounts'] = (int)$stmt->fetchColumn();
        
        $stmt = $pdo->query("SELECT SUM(amount) FROM transactions WHERE type = 'interest_credit' AND status = 'completed' AND MONTH(created_at) = MONTH(CURRENT_DATE()) AND YEAR(created_at) = YEAR(CURRENT_DATE())");
        $this_month = (float)$stmt->fetchColumn();
        $stmt = $pdo->query("SELECT SUM(amount) FROM transactions WHERE type = 'interest_credit' AND status = 'completed' AND MONTH(created_at) = MONTH(CURRENT_DATE() - INTERVAL 1 MONTH) AND YEAR(created_at) = YEAR(CURRENT_DATE() - INTERVAL 1 MONTH)");
        $last_month = (float)$stmt->fetchColumn();
        if ($last_month > 0) {
            $delta = (($this_month - $last_month) / $last_month) * 100;
            $stats['interest_delta'] = ($delta > 0 ? '+' : '') . round($delta, 1) . '%';
        }

        // Chart Data
        $chart_data = [];
        for ($i = 11; $i >= 0; $i--) {
            $stmt = $pdo->query("SELECT SUM(amount) FROM transactions WHERE type = 'interest_credit' AND status = 'completed' AND MONTH(created_at) = MONTH(CURRENT_DATE() - INTERVAL {$i} MONTH) AND YEAR(created_at) = YEAR(CURRENT_DATE() - INTERVAL {$i} MONTH)");
            $chart_data[] = (float)$stmt->fetchColumn();
        }

        // Rate change history
        $stmt = $pdo->query("
            SELECT ir.id, ir.rate, ir.apply_from_type, ir.apply_from_date, ir.created_at,
                   au.full_name as changed_by_name
            FROM interest_rates ir
            JOIN admin_users au ON ir.changed_by_admin_id = au.admin_id
            ORDER BY ir.created_at DESC
            LIMIT 10
        ");
        $history_rows = $stmt->fetchAll();
        $history = [];
        $first = true;
        foreach ($history_rows as $row) {
            $status = 'Past';
            if ($row['apply_from_date'] > date('Y-m-d')) {
                $status = 'Upcoming';
            } elseif ($first) {
                $status = 'Active';
                $first = false;
            }
            $history[] = [
                'effective_date' => date('d M Y', strtotime($row['apply_from_date'] ?? $row['created_at'])),
                'rate' => (float)$row['rate'],
                'updated_by' => $row['changed_by_name'],
                'status' => $status
            ];
        }

        json_response([
            'success'          => true,
            'current_rate'     => $current_rate,
            'last_change_date' => $last_change_date,
            'stats'            => $stats,
            'chart_data'       => $chart_data,
            'history'          => $history
        ]);

    } catch (PDOException $e) {
        error_log("Interest rate GET error: " . $e->getMessage());
        json_response(['success' => false, 'message' => 'Failed to load interest rate data'], 500);
    }
}

// =====================================================
// POST: UPDATE RATE
// =====================================================
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    check_admin_permission($pdo, $admin_id, 'change_interest_rate');

    $action = clean_input($_POST['action'] ?? 'update_rate');

    $new_rate       = floatval($_POST['new_rate'] ?? 0);
    $apply_from     = clean_input($_POST['apply_from'] ?? 'Immediately');
    $custom_date    = clean_input($_POST['custom_date'] ?? '');

    $errors = [];

    // Validate rate
    if ($new_rate < 0.01 || $new_rate > 99.99) {
        $errors['new_rate'] = 'Interest rate must be between 0.01% and 99.99%';
    }

    // Validate apply_from
    $valid_types = ['Immediately', 'Next Month', 'Custom Date'];
    if (!in_array($apply_from, $valid_types)) {
        $errors['apply_from'] = 'Invalid application type';
    }

    // Custom date required if type is Custom Date
    $apply_date = null;
    if ($apply_from === 'Custom Date') {
        if (empty($custom_date)) {
            $errors['custom_date'] = 'Custom date is required';
        } else {
            $apply_date = $custom_date;
        }
    } elseif ($apply_from === 'Next Month') {
        $apply_date = date('Y-m-01', strtotime('first day of next month'));
    } else {
        $apply_date = date('Y-m-d');
    }

    if (!empty($errors)) {
        json_response(['success' => false, 'errors' => $errors], 422);
    }

    try {
        // Fetch display admin_id (e.g. ADM001) from admin_users to prevent foreign key constraint violation
        $stmt = $pdo->prepare("SELECT admin_id FROM admin_users WHERE id = ?");
        $stmt->execute([$admin_id]);
        $admin_display_id = $stmt->fetchColumn();

        if (!$admin_display_id) {
            json_response(['success' => false, 'message' => 'Admin account not found'], 404);
        }

        // Insert new rate record
        $stmt = $pdo->prepare("
            INSERT INTO interest_rates (rate, apply_from_type, apply_from_date, changed_by_admin_id, created_at)
            VALUES (?, ?, ?, ?, NOW())
        ");
        $stmt->execute([$new_rate, $apply_from, $apply_date, $admin_display_id]);

        // If apply immediately, update all active accounts and credit the interest!
        if ($apply_from === 'Immediately') {
            // 1. Update the interest rate for all active savings accounts
            $stmt = $pdo->prepare("UPDATE accounts SET interest_rate = ?, updated_at = NOW() WHERE status = 'active' AND account_type = 'savings'");
            $stmt->execute([$new_rate]);

            // 2. Fetch all active savings accounts with a balance > 0 to credit interest
            $stmt = $pdo->query("SELECT id, user_id, balance FROM accounts WHERE account_type = 'savings' AND status = 'active' AND balance > 0");
            $savings_accounts = $stmt->fetchAll(PDO::FETCH_ASSOC);

            foreach ($savings_accounts as $acc) {
                $balance = (float)$acc['balance'];
                // Interest = Balance * (Rate / 100) / 12 months
                $interest_amount = round($balance * ($new_rate / 100) / 12, 2);

                if ($interest_amount >= 0.01) {
                    $new_balance = $balance + $interest_amount;
                    
                    // Generate unique transaction ID
                    $txn_id = 'TXN' . strtoupper(bin2hex(random_bytes(8)));

                    // Insert transaction record
                    $txStmt = $pdo->prepare("
                        INSERT INTO transactions (txn_id, from_account_id, to_account_id, type, amount, balance_after, description, reference_note, status, created_at)
                        VALUES (?, NULL, ?, 'interest_credit', ?, ?, 'Savings Interest Credit', ?, 'completed', NOW())
                    ");
                    $txStmt->execute([
                        $txn_id,
                        $acc['id'],
                        $interest_amount,
                        $new_balance,
                        "Interest credited at " . number_format($new_rate, 2) . "% p.a."
                    ]);

                    // Update the account balance
                    $upStmt = $pdo->prepare("UPDATE accounts SET balance = ?, updated_at = NOW() WHERE id = ?");
                    $upStmt->execute([$new_balance, $acc['id']]);

                    // Send notification to the user
                    $notifStmt = $pdo->prepare("
                        INSERT INTO notifications (user_id, type, title, message, is_read, created_at)
                        VALUES (?, 'transaction', 'Interest Credited', ?, 0, NOW())
                    ");
                    $notifMessage = "₹" . number_format($interest_amount, 2) . " interest has been credited to your savings account. New balance: ₹" . number_format($new_balance, 2);
                    $notifStmt->execute([$acc['user_id'], $notifMessage]);
                }
            }
        }

        // Audit log
        write_audit_log($pdo, 'interest_rate_updated', null, 'system', $admin_id, 'admin',
            "Interest rate changed to {$new_rate}%. Apply from: {$apply_from}" . ($apply_date ? " ({$apply_date})" : ''));

    $temp_pass .= '23456789'[rand(0, 7)];
    $temp_pass .= '@#$%!'[rand(0, 4)];
    for ($i = 4; $i < 12; $i++) {
        $temp_pass .= $chars[rand(0, strlen($chars) - 1)];
    }
    $temp_pass = str_shuffle($temp_pass);

    // Verify user exists and is locked
    $stmt = $pdo->prepare("SELECT id, full_name, email, failed_attempts, locked_until FROM users WHERE id = ?");
    $stmt->execute([$user_id]);
    $user = $stmt->fetch();

    if (!$user) {
        json_response(['success' => false, 'message' => 'User not found'], 404);
    }

    // Hash the temporary password
    $password_hash = password_hash($temp_pass, PASSWORD_BCRYPT);

    // Unlock the account and update password
    $stmt = $pdo->prepare("
        UPDATE users 
        SET failed_attempts = 0, 
            locked_until = NULL, 
            password_hash = ?, 
            temp_pass_status = 'done',
            updated_at = NOW() 
        WHERE id = ?
    ");
    $stmt->execute([$password_hash, $user_id]);

    // Notification for user
    send_notification($pdo, $user_id, 'security', 'Account Unlocked',
        'Your account has been unlocked by an administrator. Your password has been reset to a temporary one. Please check with an admin if you did not request this.');

    // Audit log
    write_audit_log($pdo, 'account_unlocked', $user_id, 'user', $admin_id, 'admin',
        "Manually unlocked account and reset password for {$user['full_name']} ({$user['email']}). Previous attempts: {$user['failed_attempts']}");

    // Store the temporary password in session for 5 minutes so the admin can view it if they refresh
    if (!isset($_SESSION['temp_passwords'])) {
        $_SESSION['temp_passwords'] = [];
    }
    $_SESSION['temp_passwords'][$user_id] = [
        'password' => $temp_pass,
        'expires'  => time() + 300 // 5 minutes
    ];

    json_response([
        'success' => true, 
        'message' => 'Account unlocked and temporary password generated.',
        'temp_password' => $temp_pass
    ]);

} catch (PDOException $e) {
    error_log("Unlock account error: " . $e->getMessage());
    json_response(['success' => false, 'message' => 'Failed to unlock account'], 500);
}

<?php
/**
 * Finzo Banking System — Admin User Action API
 * POST endpoint: suspend, reinstate, or soft-delete a customer account.
 * Version 1.0 — April 2026
 */

require_once dirname(__DIR__, 3) . '/middleware/auth-admin.php';

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    json_response(['success' => false, 'message' => 'Method not allowed'], 405);
}

$user_id = intval($_POST['user_id'] ?? 0);
$action  = clean_input($_POST['action'] ?? '');
$reason  = clean_input($_POST['reason'] ?? '');

if (!$user_id || empty($action)) {
    json_response(['success' => false, 'message' => 'User ID and action are required'], 422);
}

$valid_actions = ['suspend', 'reinstate', 'delete', 'recover', 'hard_delete', 'delete_account_only'];
if (!in_array($action, $valid_actions)) {
    json_response(['success' => false, 'message' => 'Invalid action'], 422);
}

$admin_id = $_SESSION['admin_id'];

try {
    // Verify user exists
    $stmt = $pdo->prepare("SELECT id, customer_id, full_name, email, status, is_deleted FROM users WHERE id = ?");
    $stmt->execute([$user_id]);
    $user = $stmt->fetch();

    if (!$user) {
        json_response(['success' => false, 'message' => 'User not found'], 404);
    }

    // Verify admin password for sensitive actions
    if (in_array($action, ['suspend', 'delete'])) {
        $password = $_POST['password'] ?? '';
        $stmt = $pdo->prepare("SELECT password_hash FROM admin_users WHERE id = ?");
        $stmt->execute([$admin_id]);
        $admin = $stmt->fetch();
        if (!$admin || !password_verify($password, $admin['password_hash'])) {
            json_response(['success' => false, 'message' => 'Invalid administrative credentials. Access denied.'], 401);
        }
    }

    switch ($action) {
        case 'suspend':
            check_admin_permission($pdo, $admin_id, 'suspend_user');
            if ($user['status'] === 'suspended') {
                json_response(['success' => false, 'message' => 'User is already suspended'], 409);
            }

            $stmt = $pdo->prepare("UPDATE users SET status = 'suspended', updated_at = NOW() WHERE id = ?");
            $stmt->execute([$user_id]);

            // Also suspend the account
            $stmt = $pdo->prepare("UPDATE accounts SET status = 'suspended', updated_at = NOW() WHERE user_id = ?");
            $stmt->execute([$user['customer_id']]);

            // Notification
            send_notification($pdo, $user_id, 'security', 'Account Suspended',
                'Your account has been suspended. Reason: ' . ($reason ?: 'No reason provided') . '. Contact support for assistance.');

            write_audit_log($pdo, 'user_suspended', $user_id, 'user', $admin_id, 'admin',
                "User {$user['full_name']} suspended. Reason: {$reason}");

            json_response(['success' => true, 'message' => 'User suspended successfully']);
            break;

        case 'reinstate':
            check_admin_permission($pdo, $admin_id, 'reinstate_user');
            if ($user['status'] === 'active') {
                json_response(['success' => true, 'message' => 'User is already active']);
            }
            if ($user['status'] !== 'suspended') {
                json_response(['success' => false, 'message' => 'User is not suspended'], 409);
            }

            $stmt = $pdo->prepare("UPDATE users SET status = 'active', updated_at = NOW() WHERE id = ?");
            $stmt->execute([$user_id]);

            $stmt = $pdo->prepare("UPDATE accounts SET status = 'active', updated_at = NOW() WHERE user_id = ? AND status = 'suspended'");
            $stmt->execute([$user['customer_id']]);

            send_notification($pdo, $user_id, 'security', 'Account Reinstated',
                'Your account has been reinstated and is now active. You can log in and transact normally.');

            write_audit_log($pdo, 'user_reinstated', $user_id, 'user', $admin_id, 'admin',
                "User {$user['full_name']} reinstated");

            json_response(['success' => true, 'message' => 'User reinstated successfully']);
            break;

        case 'delete':
            check_admin_permission($pdo, $admin_id, 'soft_delete_user');
            if ($user['is_deleted']) {
                json_response(['success' => true, 'message' => 'User is already deleted']);
            }

            // Soft delete
            $stmt = $pdo->prepare("UPDATE users SET is_deleted = 1, deleted_at = NOW(), status = 'suspended', updated_at = NOW() WHERE id = ?");
            $stmt->execute([$user_id]);

            $stmt = $pdo->prepare("UPDATE accounts SET status = 'closed', updated_at = NOW() WHERE user_id = ?");
            $stmt->execute([$user['customer_id']]);

            write_audit_log($pdo, 'user_deleted', $user_id, 'user', $admin_id, 'admin',
                "User {$user['full_name']} soft-deleted. Reason: {$reason}");

            json_response(['success' => true, 'message' => 'User deleted successfully']);
            break;

        case 'recover':
            check_admin_permission($pdo, $admin_id, 'soft_delete_user');
            if (!$user['is_deleted']) {
                json_response(['success' => false, 'message' => 'User is not deleted'], 409);
            }

            $balance_option = clean_input($_POST['balance_option'] ?? 'full');

            // Restore from soft-delete
            $stmt = $pdo->prepare("UPDATE users SET is_deleted = 0, deleted_at = NULL, status = 'active', updated_at = NOW() WHERE id = ?");
            $stmt->execute([$user_id]);

            // Reactivate account
            if ($balance_option === 'mandatory') {
                // Fetch current account details to log a charge transaction if balance > 500
                $stmt = $pdo->prepare("SELECT id, balance FROM accounts WHERE user_id = ?");
                $stmt->execute([$user['customer_id']]);
                $account = $stmt->fetch(PDO::FETCH_ASSOC);

                if ($account && $account['balance'] > 500) {
                    $deduction_amount = $account['balance'] - 500;
                    $txn_id = 'TXN' . strtoupper(uniqid());
                    // This is a system-generated charge — excluded from daily limit calculations
                    $stmt_txn = $pdo->prepare("INSERT INTO transactions (txn_id, from_account_id, type, amount, balance_after, description, reference_note, status, created_at) VALUES (?, ?, 'withdrawal', ?, 500, 'Account Restore Charge', 'Account recovered from soft-delete', 'completed', NOW())");
                    $stmt_txn->execute([$txn_id, $account['id'], $deduction_amount]);

                    // Notify user of the balance adjustment
                    send_notification($pdo, $user_id, 'system', 'Account Restored – Balance Adjusted',
                        'Your account has been restored. As per the recovery policy, your balance has been adjusted to the mandatory minimum of ₹500.00. An amount of ₹' . number_format($deduction_amount, 2) . ' was cleared as part of the restore process. This adjustment does not count toward your daily transfer limit.');
                } else {
                    // Balance is at or below 500 — just restore, no deduction needed
                    send_notification($pdo, $user_id, 'system', 'Account Restored',
                        'Your account has been restored and is now active with your mandatory minimum balance. You can log in and continue banking.');
                }

                $stmt = $pdo->prepare("UPDATE accounts SET status = 'active', balance = LEAST(balance, 500), updated_at = NOW() WHERE user_id = ?");
            } else {
                $stmt = $pdo->prepare("UPDATE accounts SET status = 'active', updated_at = NOW() WHERE user_id = ?");
                send_notification($pdo, $user_id, 'system', 'Account Restored',
                    'Your account has been restored and is now active. Your full balance has been retained. You can log in and continue banking.');
            }
            $stmt->execute([$user['customer_id']]);

            write_audit_log($pdo, 'user_recovered', $user_id, 'user', $admin_id, 'admin',
                "User {$user['full_name']} recovered from soft-delete state" . ($balance_option === 'mandatory' ? ' with mandatory balance of 500' : ''));

            json_response(['success' => true, 'message' => 'User recovered successfully']);
            break;


        case 'hard_delete':
            check_admin_permission($pdo, $admin_id, 'soft_delete_user');
            
            // Delete account completely
            $stmt = $pdo->prepare("DELETE FROM accounts WHERE user_id = ?");
            $stmt->execute([$user['customer_id']]);
            
            $stmt = $pdo->prepare("DELETE FROM users WHERE id = ?");
            $stmt->execute([$user_id]);
            
            write_audit_log($pdo, 'user_hard_deleted', $user_id, 'user', $admin_id, 'admin',
                "User {$user['full_name']} permanently deleted.");

            json_response(['success' => true, 'message' => 'User permanently deleted']);
            break;

        case 'delete_account_only':
            check_admin_permission($pdo, $admin_id, 'soft_delete_user');
            
            // Delete account completely but keep the user
            $stmt = $pdo->prepare("DELETE FROM accounts WHERE user_id = ?");
            $stmt->execute([$user['customer_id']]);
            
            write_audit_log($pdo, 'account_hard_deleted', $user_id, 'user', $admin_id, 'admin',
                "Account for user {$user['full_name']} permanently deleted.");

            json_response(['success' => true, 'message' => 'Account permanently deleted']);
            break;
    }

} catch (PDOException $e) {
    error_log("Admin user action error: " . $e->getMessage());
    json_response(['success' => false, 'message' => 'Action failed. Please try again.'], 500);
}

$user_id = intval($_GET['user_id'] ?? $_GET['id'] ?? 0);

if (!$user_id) {

}

try {
    // Fetch user profile
    $stmt = $pdo->prepare("
        SELECT id, customer_id, full_name, email, phone, status, dob, gender, father_name,
               mother_name, marital_status, nationality, profile_picture,
               occupation, annual_income, source_of_income, street, city, state, pincode, country,

               kyc_submitted_at, rejection_reason, is_deleted, deleted_at,
               failed_attempts, locked_until, last_login_at, last_login_ip,
               created_at, updated_at
        FROM users WHERE id = ?
    ");
    $stmt->execute([$user_id]);
    $user = $stmt->fetch();

    if (!$user) {
        json_response(['success' => false, 'message' => 'User not found'], 404);
    }

    // Fetch accounts
    $stmt = $pdo->prepare("
        SELECT id, account_no, account_type, balance, interest_rate, daily_limit,
               status, opened_date, created_at
        FROM accounts WHERE user_id = ?
    ");
    $stmt->execute([$user['customer_id']]);
    $accounts = $stmt->fetchAll();

    // Fetch recent transactions (last 20) across all accounts
    $transactions = [];
    if (count($accounts) > 0) {
        $account_ids = array_column($accounts, 'id');
        $placeholders = implode(',', array_fill(0, count($account_ids), '?'));
        
        $stmt = $pdo->prepare("
            SELECT
                t.txn_id, t.type, t.amount, t.balance_after, t.description,
                t.status, t.created_at,
                COALESCE(fa.account_no, '') as from_account,
                COALESCE(ta.account_no, '') as to_account
            FROM transactions t
            LEFT JOIN accounts fa ON t.from_account_id = fa.id
            LEFT JOIN accounts ta ON t.to_account_id = ta.id
            WHERE t.from_account_id IN ($placeholders) OR t.to_account_id IN ($placeholders)
            ORDER BY t.created_at DESC
            LIMIT 20
        ");
        $params = array_merge($account_ids, $account_ids);
        $stmt->execute($params);
        $transactions = $stmt->fetchAll();
    }

    // Build ID document URLs
    $id_front_url = $user['id_front_path'] ? APP_URL . '/' . $user['id_front_path'] : null;
    $id_back_url  = $user['id_back_path'] ? APP_URL . '/' . $user['id_back_path'] : null;

<?php
/**
 * Finzo Banking System — Admin Users List API
 * GET endpoint: paginated customer list with search and filters.
 * Version 1.0 — April 2026
 */

require_once dirname(__DIR__, 3) . '/middleware/auth-admin.php';

if ($_SERVER['REQUEST_METHOD'] !== 'GET') {
    json_response(['success' => false, 'message' => 'Method not allowed'], 405);
}

try {
    $search       = clean_input($_GET['search'] ?? '');
    $status       = clean_input($_GET['status'] ?? 'all');
    $page         = max(1, intval($_GET['page'] ?? 1));
    $per_page     = min(50, max(10, intval($_GET['per_page'] ?? 20)));

    // Build query
    $where_clauses = ["u.is_deleted = 0", "u.status != 'suspended'"];
    $params = [];

    // Status filter
    if ($status !== 'all' && in_array($status, ['pending', 'under_review', 'active', 'rejected'])) {
        $where_clauses[] = "u.status = ?";
        $params[] = $status;
    }

    // Search
    if (!empty($search)) {
        $where_clauses[] = "(u.full_name LIKE ? OR u.customer_id LIKE ?)";
        $search_param = "%{$search}%";
        $params[] = $search_param;
        $params[] = $search_param;
    }

    $where_sql = implode(' AND ', $where_clauses);

    // Count total
    $count_sql = "
        SELECT COUNT(u.id)
        FROM users u
        WHERE {$where_sql}
    ";
    $stmt = $pdo->prepare($count_sql);
    $stmt->execute($params);
    $total = (int)$stmt->fetchColumn();

    $pages  = max(1, ceil($total / $per_page));
    $offset = ($page - 1) * $per_page;

    // Fetch users
    $sql = "
        SELECT
            u.id, u.customer_id, u.full_name, u.email, u.phone, u.status, u.created_at, u.last_login_at, u.profile_picture
        FROM users u
        WHERE {$where_sql}
        ORDER BY u.created_at DESC
        LIMIT {$per_page} OFFSET {$offset}
    ";

    $stmt = $pdo->prepare($sql);
    $stmt->execute($params);
    $users = $stmt->fetchAll();

    // Process profile pictures and avatars
    foreach ($users as &$u) {
        $profile_pic_url = $u['profile_picture'] ? APP_URL . '/' . $u['profile_picture'] : null;
        if (!$profile_pic_url) {
            $avatar_path = 'uploads/avatars/user_' . $u['id'] . '.jpg';
            if (file_exists(dirname(__DIR__, 3) . '/' . $avatar_path)) {
                $profile_pic_url = APP_URL . '/' . $avatar_path;
            }
        }
        $u['profile_picture_url'] = $profile_pic_url;
    }
    unset($u);

    // Stats
    $stats_stmt = $pdo->query("
        SELECT
            SUM(CASE WHEN is_deleted = 0 THEN 1 ELSE 0 END) as total_users,
            SUM(CASE WHEN is_deleted = 0 AND status = 'active' THEN 1 ELSE 0 END) as active_users,
            SUM(CASE WHEN is_deleted = 0 AND status = 'suspended' THEN 1 ELSE 0 END) as suspended_users,
            SUM(CASE WHEN is_deleted = 0 AND MONTH(created_at) = MONTH(CURDATE()) AND YEAR(created_at) = YEAR(CURDATE()) THEN 1 ELSE 0 END) as new_this_month,
            SUM(CASE WHEN is_deleted = 1 THEN 1 ELSE 0 END) as soft_deleted
        FROM users
    ");
    $stats = $stats_stmt->fetch(PDO::FETCH_ASSOC);

    json_response([
        'success'      => true,
        'users'        => $users,
        'total'        => $total,
        'pages'        => $pages,
        'current_page' => $page,
        'per_page'     => $per_page,
        'stats'        => [
            'total_users'     => (int)($stats['total_users'] ?? 0),
            'active_users'    => (int)($stats['active_users'] ?? 0),
            'suspended_users' => (int)($stats['suspended_users'] ?? 0),
            'new_this_month'  => (int)($stats['new_this_month'] ?? 0),
            'soft_deleted'    => (int)($stats['soft_deleted'] ?? 0)
        ]
    ]);

} catch (PDOException $e) {
    error_log("Admin users API error: " . $e->getMessage());
    json_response(['success' => false, 'message' => 'Failed to load users'], 500);
}
